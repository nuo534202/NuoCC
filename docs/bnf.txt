/*
 * The source has been through the C pre-processor before it is scanned, so
 * the comments, the #include and #define lines and the markers the
 * pre-processor writes about the file and line it came from are all gone.
 * What is left is the grammar below.
 */
program: global_declaration
    | global_declaration program
    ;

/*
 * A declaration may be written with one of two words in front of its type.
 * `extern` says that the storage of the name belongs to another source
 * file, so no storage is emitted for it here, and `static` that it belongs
 * to this one alone, so the assembler is not told another file may reach
 * it. The two may not be written together.
 */
global_declaration: opt_storage_class function_declaration
    | opt_storage_class var_declaration
    | typedef_declaration
    ;

opt_storage_class: <empty>
    | 'extern'
    | 'static'
    ;

function_declaration:
      type identifier '(' opt_parameter_list ')' compound_statement
    | type identifier '(' opt_parameter_list ')' ';'
    ;

/*
 * A declaration which ends in a semicolon is a prototype: it declares the
 * function and the types it takes, and the body comes later. A definition
 * which follows has to agree with the prototype it repeats.
 */

/*
 * A parameter list may also be the one word `void`, which says the same as
 * an empty list: the function takes nothing.
 */
opt_parameter_list: <empty>
    | 'void'
    | parameter_list
    | parameter_list ',' '...'
    ;

parameter_list: parameter
    | parameter ',' parameter_list
    ;

/*
 * A list which ends in '...' names the first few arguments of a function
 * which takes more of them than it names, as `printf()` does. It has to
 * name at least one parameter for the '...' to stand after, and a function
 * so declared is one written elsewhere: a body may not be given to it,
 * since the language has no way of reaching the arguments it does not name.
 */

/*
 * A parameter holds a value, so one which is given an array holds the
 * address of its first element and is written as a pointer; a '[' after
 * the name is refused rather than read, with a message saying so.
 */
parameter: type opt_pointer identifier
    ;

/*
 * Every name of a declaration brings its own stars and its own array size,
 * so plain values, pointers and arrays may stand in one list. An array
 * holds a fixed number of elements, which may not be zero, and every
 * element has to be a type which takes room, which a struct or a union
 * does. An array of composite values cannot be given values, since there is
 * no way to write a composite value, so what tells such an array apart is
 * that its elements are only ever stored into one member at a time.
 */
var_declaration: type declarator_list ';'
    ;

declarator_list: declarator
    | declarator ',' declarator_list
    ;

/*
 * A name may be given the value or the values it starts with, and an array
 * the size it holds, and each of them is an expression. A value written
 * where a global variable is declared has to be one the compiler can work
 * out while it is compiling the program, since the compiler is what stores
 * it: `int x= 10 + 6;` is the value 16, and a string is the address of its
 * characters. A value written where a local variable is declared, array or
 * not, is an expression which runs where it stands, and the values of an
 * array are stored into its elements there one after another, while an
 * array of composite values is cleared instead, since none of its elements
 * can be given anything. An array
 * written without a size takes the number of values it is given as its
 * size. A member of a struct is not a place of its own, so it may not be
 * given values at all.
 *
 * An array of char may be given a string in place of the list of values,
 * which gives it the characters of the string one element each and the zero
 * byte it ends with as one element more when it has room for it. A string
 * written for an array of any other type is refused.
 *
 * A value of a composite type cannot be read as a value, so what it is
 * given is written one of the two ways a composite value ever is: as the
 * values of the members of its type between braces, or as another place
 * holding a value of the same type, which is written as an expression and
 * is copied whole. The members are written in the order the definition
 * gives them, and a member which is a composite value of its own takes a
 * list of its own, while one which is an array takes a list of element
 * values or, when it holds characters, the characters of a string. A union
 * takes the value written for its first member. The braces may be closed
 * before every member has been written for, and what is not written is
 * left at zero.
 */
declarator: opt_pointer identifier opt_value
    | opt_pointer identifier '[' opt_size ']' opt_values
    ;

opt_value: <empty>
    | '=' expression
    | '=' '{' member_value_list '}'
    ;

member_value: expression
    | string_literal
    | '{' member_value_list '}'
    ;

member_value_list: <empty>
    | member_value
    | member_value ',' member_value_list
    ;

opt_values: <empty>
    | '=' '{' value_list '}'
    | '=' string_literal
    ;

value: expression
    ;

value_list: <empty>
    | value
    | value ',' value_list
    ;

opt_size: <empty>
    | expression
    ;

/*
 * A type is a keyword, a composite or an enum named by its tag, or a name
 * made with typedef, which stands for whatever type it was declared to be.
 * Only the base type is written here: the stars belong to the declarator,
 * which is where each name of a declaration is given its own, so that
 * `int a, *b;` declares a variable and a pointer in one declaration.
 */
type: type_keyword
    | composite_type
    | enum_type
    | identifier
    ;

type_keyword: 'char'
    | 'short'
    | 'int'
    | 'long'
    | 'void'
    ;

/*
 * A struct or a union is named by its tag, and a type which is written
 * without one can only be used where it is defined. The members are read
 * with the type already known by its tag, so that a member may be a
 * pointer back at the composite type it stands in. A type with no members
 * is refused.
 */
composite_type: struct_type
    | union_type
    ;

struct_type: 'struct' opt_tag
    | 'struct' opt_tag '{' member_list '}'
    ;

union_type: 'union' opt_tag
    | 'union' opt_tag '{' member_list '}'
    ;

/*
 * An enum shares the tag namespace with structs and unions, so one tag
 * cannot be reused as another kind. A value of an enum is an int, so the
 * type is an int however the enum was written. The names between the
 * braces are ordinary names for the numbers they stand for, counting up
 * from zero and starting again from each number written out.
 */
enum_type: 'enum' opt_tag
    | 'enum' opt_tag '{' enum_value_list '}'
    ;

enum_value_list: enum_value
    | enum_value ',' enum_value_list
    ;

enum_value: identifier
    | identifier '=' expression
    ;

opt_tag: <empty>
    | identifier
    ;

member_list: member
    | member member_list
    ;

/*
 * A member of a struct or a union may be a pointer or an array of its own,
 * but it cannot be given a value where it is declared. The size of such an
 * array is an expression the compiler works out while it is compiling the
 * program, as the size of any other array is.
 */
member: type opt_pointer identifier ';'
    | type opt_pointer identifier '[' expression ']' ';'
    ;

opt_pointer: <empty>
    | '*' opt_pointer
    ;

/*
 * A block is a statement like any other, so braces are needed where more
 * than one statement has to stand in a place which holds one.
 */
compound_statement: '{' '}'
    | '{' statement '}'
    | '{' statement statements '}'
    ;

statement: compound_statement
    | print_statement
    | declaration
    | expression_statement
    | empty_statement
    | if_statement
    | while_statement
    | do_statement
    | for_statement
    | switch_statement
    | return_statement
    | break_statement
    | continue_statement
    | goto_statement
    | label_statement
    ;

print_statement: 'print' expression ';'
    ;

/*
 * A name declared inside a function may be `extern`, but not `static`: a
 * static name belongs to the file it is written in, so it can only be
 * declared where the whole file can be seen.
 */
declaration: opt_extern type declarator_list ';'
    | typedef_declaration
    ;

/*
 * The type has already been read by the time the new name is reached, and
 * the name becomes another way of writing that type. A type definition
 * which names nothing may stand alone, as `enum fred { apple };` does.
 */
typedef_declaration: 'typedef' type opt_pointer identifier ';'
    ;

/*
 * What is called has to be the name of a function, and it is given one
 * expression for each of its parameters: at most six of them, which is how
 * many the machine passes without reaching into the stack.
 * Each expression has to fit the type of the parameter it is given to.
 * A function declared with '...' names only the first few of the arguments
 * it is given, so a call to one may give more arguments than that, and
 * every argument beyond the named ones takes the type an argument is given
 * when nothing says otherwise.
 */
function_call: identifier '(' opt_expression_list ')'
    ;

opt_expression_list: <empty>
    | expression_list
    ;

/*
 * The expressions of a call are given one after another, and the same list
 * is what stands in the first and the last clause of a for loop, where it
 * may also be left empty.
 */
expression_list: expression
    | expression ',' expression_list
    ;

/*
 * A function which answers with a value has to be given one to return, and
 * one which answers with nothing leaves with a return which is given none.
 */
return_statement: 'return' '(' expression ')'
    | 'return'
    ;

/*
 * A break leaves the loop or the switch it stands in, and a continue
 * begins that loop's next pass, so each may only be written where there is
 * something to leave or to go on with.
 */
break_statement: 'break' ';'
    ;

continue_statement: 'continue' ';'
    ;

/*
 * A goto jumps to a label of the function it stands in, which may be
 * written after it, since the whole of a function is read before any of it
 * is made into code. A label names the statement which follows it, which
 * may be a ';' on its own and so nothing at all, and no two labels of one
 * function may have the same name.
 */
goto_statement: 'goto' identifier ';'
    ;

label_statement: identifier ':' statement
    ;

expression_statement: expression ';'
    ;

/* A ';' on its own is a statement which does nothing, which is how a loop
 * or a branch with nothing to do is written. */
empty_statement: ';'
    ;

/*
 * The else is taken by the if it stands nearest, because the branch of an
 * if is parsed as far as it goes before an else is looked for.
 */
if_statement: if_head
    | if_head 'else' statement
    ;

if_head: 'if' '(' true_false_expression ')' statement
    ;

while_statement: 'while' '(' true_false_expression ')' statement
    ;

/*
 * The body of a do loop runs once before the condition is asked at all, so
 * the condition stands after it and the loop goes round again only where
 * it held. A continue in the body goes on to the condition, since the
 * condition is what decides whether there is another pass.
 */
do_statement: 'do' statement 'while' '(' true_false_expression ')' ';'
    ;

/*
 * The first and the last clause are lists of expressions which run in the
 * order they are written, as in `for (x= 0, y= 1; x < 6; x++, y= y + 2)`,
 * and either may be left empty. The step between the last two semicolons
 * runs at the end of every pass, and is where a continue in the body goes
 * on to, so it runs on that pass as well.
 */
for_statement: 'for' '(' expression_list ';'
                       true_false_expression ';'
                       expression_list ')' statement
    ;

/*
 * A switch tries its arms in the order they are written and takes the
 * first whose value matches, and the default arm when none does. An arm
 * which does not end in a break falls through into the arm after it, so
 * the arms are laid out one after another, and the default arm has to be
 * the last one. A case value is an integer constant: a literal, or the
 * name an enum gave to one.
 */
switch_statement: 'switch' '(' expression ')' '{' case_list '}'
    ;

case_list: case_item
    | case_item case_list
    ;

/*
 * What a case names has to be a number the compiler can work out, so a
 * name which stands for one, and an expression of them, are both taken.
 */
case_item: 'case' expression ':' case_body
    | 'default' ':' case_body
    ;

/*
 * The statements of an arm run one after another, until the next case, the
 * default or the end of the switch. More than one of them may stand there
 * without braces, and a '{' simply begins a block like it does anywhere.
 */
case_body: <empty>
    | statement case_body
    ;

identifier: T_IDENT
    ;

/*
 * The operators, from the loosest to the tightest. '=' binds less tightly
 * than every other one and to the right, so that a= b= 3 stores into b
 * first and then into a. Only a variable and the location a pointer holds
 * can stand on its left.
 */
expression: assignment_expression
    ;

/*
 * Every assignment takes what stands on its right first, so they all bind
 * to the right. `=` stores the value as it stands, while every other one
 * combines it with the value the target holds: `+=`, `-=`, `*=` and `/=`
 * arithmetically, `%=` by the remainder, and `&=`, `|=`, `^=`, `<<=` and
 * `>>=` bitwise or by a shift.
 *
 * An assignment whose target holds a composite value stores no value: it
 * gives the room of the target the contents of another place holding the
 * same type. The right side is therefore a place and not a value, and only
 * `=` is read for one, since there is no operation which combines two
 * composite values.
 */
assignment_expression: conditional_expression
    | lvalue '=' assignment_expression
    | lvalue '+=' assignment_expression
    | lvalue '-=' assignment_expression
    | lvalue '*=' assignment_expression
    | lvalue '/=' assignment_expression
    | lvalue '%=' assignment_expression
    | lvalue '&=' assignment_expression
    | lvalue '|=' assignment_expression
    | lvalue '^=' assignment_expression
    | lvalue '<<=' assignment_expression
    | lvalue '>>=' assignment_expression
    ;

/*
 * The condition is worked out first and only the value it chooses is worked
 * out at all: `x= 1 ? 5 : counted();` never calls counted(). The two values
 * have to be of one type, and an expression whose two values answer with
 * nothing is one of those itself. The operator binds more loosely than
 * every operator but the assignments, and to the right, so the '?' of
 * `a ? b : c ? d : e` belongs to the value taken when a does not hold.
 */
conditional_expression: logical_or_expression
    | logical_or_expression '?' expression ':' expression
    ;

lvalue: identifier
    | array_access
    | member_access
    | '*' prefix_expression
    ;

logical_or_expression:
      logical_and_expression
    | logical_or_expression '||' logical_and_expression
    ;

logical_and_expression:
      or_expression
    | logical_and_expression '&&' or_expression
    ;

or_expression:
      xor_expression
    | or_expression '|' xor_expression
    ;

xor_expression:
      amper_expression
    | xor_expression '^' amper_expression
    ;

amper_expression:
      equality_expression
    | amper_expression '&' equality_expression
    ;

/*
 * A condition is any expression. A comparison already says whether it
 * holds; any other value holds when it is not zero.
 */
true_false_expression: expression
    ;

equality_expression:
      relational_expression
    | equality_expression '==' relational_expression
    | equality_expression '!=' relational_expression
    ;

relational_expression:
      shift_expression
    | relational_expression '<' shift_expression
    | relational_expression '>' shift_expression
    | relational_expression '<=' shift_expression
    | relational_expression '>=' shift_expression
    ;

shift_expression:
      additive_expression
    | shift_expression '<<' additive_expression
    | shift_expression '>>' additive_expression
    ;

additive_expression:
      multiplicative_expression
    | additive_expression '+' multiplicative_expression
    | additive_expression '-' multiplicative_expression
    ;

multiplicative_expression:
      prefix_expression
    | multiplicative_expression '*' prefix_expression
    | multiplicative_expression '/' prefix_expression
    | multiplicative_expression '%' prefix_expression
    ;

/*
 * The prefix operators apply to whatever they are written before, and '++'
 * and '--' also apply to a place they are written after. An increment or a
 * decrement has to have somewhere to put the new value, so what it is
 * written on is a place: the name of a variable, a place a pointer reaches,
 * or a member of a value.
 */
prefix_expression: postfix_expression
    | cast_expression
    | '*' prefix_expression
    | '&' prefix_expression
    | '-' prefix_expression
    | '~' prefix_expression
    | '!' prefix_expression
    | '++' prefix_expression
    | '--' prefix_expression
    ;

postfix_expression: primary
    | array_access
    | member_access
    | postfix_expression '++'
    | postfix_expression '--'
    ;

/*
 * A cast names a type between parentheses and holds on to the one
 * expression which follows it, the way a unary operator does. The type may
 * not be a struct, a union or a plain void, since none of them holds a
 * value, but a pointer to void is a value like any other.
 */
cast_expression: '(' type opt_pointer ')' prefix_expression
    ;

/*
 * An array named on its own stands for the address of its first element,
 * so an array_access may also be written where a pointer is wanted, and
 * the index counts elements rather than bytes. An element which is a
 * composite value takes the room its own definition asks for, so the same
 * index reaches the same element whether the room of one is the room of a
 * member or the room of a struct.
 */
array_access: postfix_expression '[' expression ']'
    ;

/* Members select a struct value or a pointer to one and may be chained. */
member_access: postfix_expression '.' identifier
    | postfix_expression '->' identifier
    ;

primary: number
    | string_literal
    | identifier
    | function_call
    | sizeof_expression
    | '(' expression ')'
    ;

/*
 * sizeof answers with the room a value takes, in bytes. What stands between
 * the parentheses is a type when it begins like one and a value otherwise,
 * the same way a cast is told from a parenthesised expression. A value
 * written there says which type is asked about and is not worked out, and
 * an array named on its own asks about the room the whole array takes.
 */
sizeof_expression: 'sizeof' '(' type opt_pointer ')'
    | 'sizeof' '(' expression ')'
    ;

/*
 * A character literal scans as a number whose value is the character's
 * code and which the parser gives the type char when it fits, so it is
 * not a production of its own. A string literal is a token of its own and
 * has the type char *.
 *
 * A number is written in the radix it says it is written in: ten, or
 * eight when it begins with a zero, or sixteen when it begins with 0x.
 * A character may also be named by its code, in octal or in hexadecimal,
 * inside a character or a string literal.
 */
number: T_INTLIT
    ;

/*
 * Strings written one after another are one string, which is how a string
 * too long for its line is written down over several.
 */
string_literal: T_STRLIT
    | string_literal T_STRLIT
    ;
