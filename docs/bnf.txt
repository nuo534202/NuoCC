program: global_declaration
    | global_declaration program
    ;

global_declaration: function_declaration
    | var_declaration
    ;

function_declaration: type identifier '(' opt_parameter ')' compound_statement
    ;

opt_parameter: <empty>
    | type identifier
    ;

var_declaration: type identifier_list ';'
    ;

identifier_list: identifier
    | identifier ',' identifier_list
    ;

type: type_keyword opt_pointer
    ;

type_keyword: 'char'
    | 'int'
    | 'long'
    | 'void'
    ;

opt_pointer: <empty>
    | '*' opt_pointer
    ;

compound_statement:
      '{' '}'
    | '{' statement '}'
    | '{' statement statements '}'
    ;

statement: print_statement
    | declaration
    | assignment_statement
    | function_call
    | if_statement
    | while_statement
    | for_statement
    | return_statement
    ;

print_statement: 'print' expression ';'
    ;

declaration: type identifier_list ';'
    ;

function_call: identifier '(' opt_expression ')'
    ;

opt_expression: <empty>
    | expression
    ;

return_statement: 'return' '(' expression ')'
    ;

assignment_statement: identifier '=' expression ';'
    ;

if_statement: if_head
    | if_head 'else' compound_statement
    ;

if_head: 'if' '(' true_false_expression ')' compound_statement
    ;

while_statement: 'while' '(' true_false_expression ')' compound_statement
    ;

for_statement: 'for' '(' statement ';'
                       true_false_expression ';'
                       statement ')' compound_statement
    ;

identifier: T_IDENT
    ;

true_false_expression: equality_expression
    ;

equality_expression:
      relational_expression
    | equality_expression '==' relational_expression
    | equality_expression '!=' relational_expression
    ;

relational_expression:
      additive_expression
    | relational_expression '<' additive_expression
    | relational_expression '>' additive_expression
    | relational_expression '<=' additive_expression
    | relational_expression '>=' additive_expression
    ;

additive_expression:
      multiplicative_expression
    | additive_expression '+' multiplicative_expression
    | additive_expression '-' multiplicative_expression
    ;

multiplicative_expression:
      prefix_expression
    | multiplicative_expression '*' prefix_expression
    | multiplicative_expression '/' prefix_expression
    ;

primary: number
    | identifier
    | function_call
    ;

prefix_expression: primary
    | '*' prefix_expression
    | '&' prefix_expression
    ;

number: T_INTLIT
    ;
