/*
 * The source has been through the C pre-processor before it is scanned, so
 * the comments, the #include and #define lines and the markers the
 * pre-processor writes about the file and line it came from are all gone.
 * What is left is the grammar below.
 */
program: global_declaration
    | global_declaration program
    ;

/*
 * A declaration may be written `extern`, which says that the storage of
 * what it declares is given in another source file. It has to come before
 * the type, and no storage is emitted for the names it declares.
 */
global_declaration: opt_extern function_declaration
    | opt_extern var_declaration
    | typedef_declaration
    ;

opt_extern: <empty>
    | 'extern'
    ;

function_declaration:
      type identifier '(' opt_parameter_list ')' compound_statement
    | type identifier '(' opt_parameter_list ')' ';'
    ;

/*
 * A declaration which ends in a semicolon is a prototype: it declares the
 * function and the types it takes, and the body comes later. A definition
 * which follows has to agree with the prototype it repeats.
 */

opt_parameter_list: <empty>
    | parameter_list
    ;

parameter_list: parameter
    | parameter ',' parameter_list
    ;

parameter: type opt_pointer identifier
    ;

/*
 * Every name of a declaration brings its own stars and its own array size,
 * so plain values, pointers and arrays may stand in one list. An array
 * holds a fixed number of elements, which may not be zero, and every
 * element has to be a type which can hold a value.
 */
var_declaration: type declarator_list ';'
    ;

declarator_list: declarator
    | declarator ',' declarator_list
    ;

declarator: opt_pointer identifier
    | opt_pointer identifier '[' number ']'
    ;

/*
 * A type is a keyword, a composite or an enum named by its tag, or a name
 * made with typedef, which stands for whatever type it was declared to be.
 * Only the base type is written here: the stars belong to the declarator,
 * which is where each name of a declaration is given its own, so that
 * `int a, *b;` declares a variable and a pointer in one declaration.
 */
type: type_keyword
    | composite_type
    | enum_type
    | identifier
    ;

type_keyword: 'char'
    | 'int'
    | 'long'
    | 'void'
    ;

/*
 * A struct or a union is named by its tag, and a type which is written
 * without one can only be used where it is defined. The members are read
 * with the type already known by its tag, so that a member may be a
 * pointer back at the composite type it stands in. A type with no members
 * is refused.
 */
composite_type: struct_type
    | union_type
    ;

struct_type: 'struct' opt_tag
    | 'struct' opt_tag '{' member_list '}'
    ;

union_type: 'union' opt_tag
    | 'union' opt_tag '{' member_list '}'
    ;

/*
 * An enum shares the tag namespace with structs and unions, so one tag
 * cannot be reused as another kind. A value of an enum is an int, so the
 * type is an int however the enum was written. The names between the
 * braces are ordinary names for the numbers they stand for, counting up
 * from zero and starting again from each number written out.
 */
enum_type: 'enum' opt_tag
    | 'enum' opt_tag '{' enum_value_list '}'
    ;

enum_value_list: enum_value
    | enum_value ',' enum_value_list
    ;

enum_value: identifier
    | identifier '=' number
    ;

opt_tag: <empty>
    | identifier
    ;

member_list: member
    | member member_list
    ;

member: type declarator ';'
    ;

opt_pointer: <empty>
    | '*' opt_pointer
    ;

/*
 * A block is a statement like any other, so braces are needed where more
 * than one statement has to stand in a place which holds one.
 */
compound_statement: '{' '}'
    | '{' statement '}'
    | '{' statement statements '}'
    ;

statement: compound_statement
    | print_statement
    | declaration
    | expression_statement
    | if_statement
    | while_statement
    | for_statement
    | switch_statement
    | return_statement
    | break_statement
    | continue_statement
    ;

print_statement: 'print' expression ';'
    ;

declaration: opt_extern type declarator_list ';'
    | typedef_declaration
    ;

/*
 * The type has already been read by the time the new name is reached, and
 * the name becomes another way of writing that type. A type definition
 * which names nothing may stand alone, as `enum fred { apple };` does.
 */
typedef_declaration: 'typedef' type opt_pointer identifier ';'
    ;

function_call: identifier '(' opt_expression_list ')'
    ;

opt_expression_list: <empty>
    | expression_list
    ;

/*
 * The expressions of a call are given one after another, and the same list
 * is what stands in the first and the last clause of a for loop, where it
 * may also be left empty.
 */
expression_list: expression
    | expression ',' expression_list
    ;

return_statement: 'return' '(' expression ')'
    ;

/*
 * A break leaves the loop or the switch it stands in, and a continue
 * begins that loop's next pass, so each may only be written where there is
 * something to leave or to go on with.
 */
break_statement: 'break' ';'
    ;

continue_statement: 'continue' ';'
    ;

expression_statement: expression ';'
    ;

/*
 * The else is taken by the if it stands nearest, because the branch of an
 * if is parsed as far as it goes before an else is looked for.
 */
if_statement: if_head
    | if_head 'else' statement
    ;

if_head: 'if' '(' true_false_expression ')' statement
    ;

while_statement: 'while' '(' true_false_expression ')' statement
    ;

/*
 * The first and the last clause are lists of expressions which run in the
 * order they are written, as in `for (x= 0, y= 1; x < 6; x++, y= y + 2)`,
 * and either may be left empty. The step between the last two semicolons
 * runs at the end of every pass, and is where a continue in the body goes
 * on to, so it runs on that pass as well.
 */
for_statement: 'for' '(' expression_list ';'
                       true_false_expression ';'
                       expression_list ')' statement
    ;

/*
 * A switch tries its arms in the order they are written and takes the
 * first whose value matches, and the default arm when none does. An arm
 * which does not end in a break falls through into the arm after it, so
 * the arms are laid out one after another, and the default arm has to be
 * the last one. A case value is an integer constant: a literal, or the
 * name an enum gave to one.
 */
switch_statement: 'switch' '(' expression ')' '{' case_list '}'
    ;

case_list: case_item
    | case_item case_list
    ;

case_item: 'case' number ':' case_body
    | 'default' ':' case_body
    ;

/*
 * The statements of an arm run one after another, until the next case, the
 * default or the end of the switch. More than one of them may stand there
 * without braces, and a '{' simply begins a block like it does anywhere.
 */
case_body: <empty>
    | statement case_body
    ;

identifier: T_IDENT
    ;

/*
 * The operators, from the loosest to the tightest. '=' binds less tightly
 * than every other one and to the right, so that a= b= 3 stores into b
 * first and then into a. Only a variable and the location a pointer holds
 * can stand on its left.
 */
expression: assignment_expression
    ;

assignment_expression: logical_or_expression
    | lvalue '=' assignment_expression
    ;

lvalue: identifier
    | array_access
    | member_access
    | '*' prefix_expression
    ;

logical_or_expression:
      logical_and_expression
    | logical_or_expression '||' logical_and_expression
    ;

logical_and_expression:
      or_expression
    | logical_and_expression '&&' or_expression
    ;

or_expression:
      xor_expression
    | or_expression '|' xor_expression
    ;

xor_expression:
      amper_expression
    | xor_expression '^' amper_expression
    ;

amper_expression:
      equality_expression
    | amper_expression '&' equality_expression
    ;

/*
 * A condition is any expression. A comparison already says whether it
 * holds; any other value holds when it is not zero.
 */
true_false_expression: expression
    ;

equality_expression:
      relational_expression
    | equality_expression '==' relational_expression
    | equality_expression '!=' relational_expression
    ;

relational_expression:
      shift_expression
    | relational_expression '<' shift_expression
    | relational_expression '>' shift_expression
    | relational_expression '<=' shift_expression
    | relational_expression '>=' shift_expression
    ;

shift_expression:
      additive_expression
    | shift_expression '<<' additive_expression
    | shift_expression '>>' additive_expression
    ;

additive_expression:
      multiplicative_expression
    | additive_expression '+' multiplicative_expression
    | additive_expression '-' multiplicative_expression
    ;

multiplicative_expression:
      prefix_expression
    | multiplicative_expression '*' prefix_expression
    | multiplicative_expression '/' prefix_expression
    ;

/*
 * The prefix operators apply to whatever they are written before, and '++'
 * and '--' also apply to a name they are written after. An increment or a
 * decrement has to have somewhere to put the new value, so it only takes
 * the name of a variable.
 */
prefix_expression: postfix_expression
    | '*' prefix_expression
    | '&' prefix_expression
    | '-' prefix_expression
    | '~' prefix_expression
    | '!' prefix_expression
    | '++' prefix_expression
    | '--' prefix_expression
    ;

postfix_expression: primary
    | array_access
    | member_access
    | identifier '++'
    | identifier '--'
    ;

/*
 * An array named on its own stands for the address of its first element,
 * so an array_access may also be written where a pointer is wanted, and
 * the index counts elements rather than bytes.
 */
array_access: postfix_expression '[' expression ']'
    ;

/* Members select a struct value or a pointer to one and may be chained. */
member_access: postfix_expression '.' identifier
    | postfix_expression '->' identifier
    ;

primary: number
    | string_literal
    | identifier
    | function_call
    | '(' expression ')'
    ;

/*
 * A character literal scans as a number whose value is the character's
 * code and which the parser gives the type char when it fits, so it is
 * not a production of its own. A string literal is a token of its own
 * and has the type char *.
 */
number: T_INTLIT
    ;

string_literal: T_STRLIT
    ;
